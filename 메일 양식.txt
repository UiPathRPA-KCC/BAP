Dim sb As New System.Text.StringBuilder()

Dim hiddenCols As New List(Of String) From {"ArticleURL"}

'---------------------------------------
' HTML 시작
'---------------------------------------
sb.AppendLine("<html><body style='font-family:Segoe UI, Arial; font-size:14px;'>")
sb.AppendLine("<h3 style='margin:0 0 8px 0;'>기사 리포트</h3>")

'=========================================================
' 1) DT_Scraping
'=========================================================
sb.AppendLine("<h4 style='margin:14px 0 6px 0;'>1) 오늘의 인기 IT 기사</h4>")

If in_DT_Scraping Is Nothing OrElse in_DT_Scraping.Rows.Count = 0 Then
    sb.AppendLine("<p>결과가 없습니다.</p>")
Else
    Dim hasTitle1 As Boolean = in_DT_Scraping.Columns.Contains("ArticleTitle")
    Dim hasUrl1 As Boolean = in_DT_Scraping.Columns.Contains("ArticleURL")

    sb.AppendLine("<table border='1' cellpadding='6' cellspacing='0' style='border-collapse:collapse; width:100%; font-size:13px;'>")

    '헤더
    sb.AppendLine("<tr style='background:#f2f2f2;'>")
    For Each col As System.Data.DataColumn In in_DT_Scraping.Columns
        If Not hiddenCols.Contains(col.ColumnName) Then
            sb.AppendLine("<th>" & System.Net.WebUtility.HtmlEncode(col.ColumnName) & "</th>")
        End If
    Next
    sb.AppendLine("</tr>")

    '데이터
    For Each r As System.Data.DataRow In in_DT_Scraping.Rows
        sb.AppendLine("<tr>")

        For Each col As System.Data.DataColumn In in_DT_Scraping.Columns
            Dim colName As String = col.ColumnName
            If hiddenCols.Contains(colName) Then
                Continue For
            End If

            Dim cell As String = If(r.IsNull(colName), "", r(colName).ToString())

            '✅ 제목 컬럼이면 링크로 출력(표에는 URL 컬럼 표시 안 함)
            If hasTitle1 AndAlso hasUrl1 AndAlso colName = "ArticleTitle" Then
                Dim link As String = If(r.IsNull("ArticleURL"), "", r("ArticleURL").ToString())
                Dim titleEnc As String = System.Net.WebUtility.HtmlEncode(cell)
                Dim linkSafe As String = link.Replace("'", "%27")

                If Not String.IsNullOrWhiteSpace(linkSafe) Then
                    sb.AppendLine("<td><a href='" & linkSafe & "' target='_blank'>" & titleEnc & "</a></td>")
                Else
                    sb.AppendLine("<td>" & titleEnc & "</td>")
                End If
            Else
                sb.AppendLine("<td>" & System.Net.WebUtility.HtmlEncode(cell) & "</td>")
            End If
        Next

        sb.AppendLine("</tr>")
    Next

    sb.AppendLine("</table>")
End If

'=========================================================
' 2) DT_Keyword
'=========================================================
sb.AppendLine("<h4 style='margin:18px 0 6px 0;'>2) 키워드 검색 결과</h4>")

If in_DT_Keyword Is Nothing OrElse in_DT_Keyword.Rows.Count = 0 Then
    sb.AppendLine("<p>결과가 없습니다.</p>")
Else
    Dim hasTitle2 As Boolean = in_DT_Keyword.Columns.Contains("ArticleTitle")
    Dim hasUrl2 As Boolean = in_DT_Keyword.Columns.Contains("ArticleURL")

    sb.AppendLine("<table border='1' cellpadding='6' cellspacing='0' style='border-collapse:collapse; width:100%; font-size:13px;'>")

    '헤더
    sb.AppendLine("<tr style='background:#f2f2f2;'>")
    For Each col As System.Data.DataColumn In in_DT_Keyword.Columns
        If Not hiddenCols.Contains(col.ColumnName) Then
            sb.AppendLine("<th>" & System.Net.WebUtility.HtmlEncode(col.ColumnName) & "</th>")
        End If
    Next
    sb.AppendLine("</tr>")

    '데이터
    For Each r As System.Data.DataRow In in_DT_Keyword.Rows
        sb.AppendLine("<tr>")

        For Each col As System.Data.DataColumn In in_DT_Keyword.Columns
            Dim colName As String = col.ColumnName
            If hiddenCols.Contains(colName) Then
                Continue For
            End If

            Dim cell As String = If(r.IsNull(colName), "", r(colName).ToString())

            If hasTitle2 AndAlso hasUrl2 AndAlso colName = "ArticleTitle" Then
                Dim link As String = If(r.IsNull("ArticleURL"), "", r("ArticleURL").ToString())
                Dim titleEnc As String = System.Net.WebUtility.HtmlEncode(cell)
                Dim linkSafe As String = link.Replace("'", "%27")

                If Not String.IsNullOrWhiteSpace(linkSafe) Then
                    sb.AppendLine("<td><a href='" & linkSafe & "' target='_blank'>" & titleEnc & "</a></td>")
                Else
                    sb.AppendLine("<td>" & titleEnc & "</td>")
                End If
            Else
                sb.AppendLine("<td>" & System.Net.WebUtility.HtmlEncode(cell) & "</td>")
            End If
        Next

        sb.AppendLine("</tr>")
    Next

    sb.AppendLine("</table>")
End If

'---------------------------------------
' HTML 끝
'---------------------------------------
sb.AppendLine("</body></html>")

out_MailBodyHtml = sb.ToString()
